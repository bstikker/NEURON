% ---------- Building blocks ----------
% Small grey label above a value
\newcommand{\fieldlabel}[1]{{\scriptsize\color{muted}\MakeUppercase{#1}}}

% Key-result card: \resultcard{label}{headline}{detail}
\newtcolorbox{cardbox}{enhanced, colback=white, colframe=rule, boxrule=0.6pt, arc=2pt,
  borderline north={2.5pt}{0pt}{accent}, left=9pt, right=9pt, top=8pt, bottom=8pt}
\newcommand{\resultcard}[3]{%
  \begin{cardbox}
    \fieldlabel{#1}\par\vspace{2pt}
    {\large\bfseries\color{brand}#2\par}\vspace{3pt}
    {\small\color{muted}#3\par}
  \end{cardbox}}

% Light panel for the sample details strip
